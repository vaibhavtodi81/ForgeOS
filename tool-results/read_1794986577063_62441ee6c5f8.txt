   208→  /** Splits totalQty into n dated OUT chunks spread over the last `span` days. */
   209→  async function addOuts(projectId: string, materialName: string, totalQty: number, chunks: number, spanDays: number, recorderId: string) {
   210→    const per = totalQty / chunks
   211→    for (let i = 0; i < chunks; i++) {
   212→      const daysOld = Math.round(spanDays * (1 - (i + 0.5) / chunks))
   213→      await db.inventoryTransaction.create({
   214→        data: {
   215→          projectId, materialId: mats[materialName], txnType: 'OUT',
   216→          quantity: Math.round(jitter(per, 0.08) * 100) / 100, unitRate: 0,
   217→          txnDate: daysAgo(daysOld), recordedById: recorderId, createdAt: daysAgo(daysOld, 17),
   218→        },
   219→      })
   220→    }
   221→  }
