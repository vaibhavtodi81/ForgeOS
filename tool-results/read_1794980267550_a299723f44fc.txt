   362→function PaymentsDialog({
   363→  invoice,
   364→  isFinance,
   365→  onClose,
   366→}: {
   367→  invoice: InvoiceRow
   368→  isFinance: boolean
   369→  onClose: () => void
   370→}) {
   371→  const queryClient = useQueryClient()
   372→
   373→  const paymentsQuery = useQuery({
   374→    queryKey: ['payments', invoice.id],
   375→    queryFn: () =>
   376→      api<{ items: PaymentRow[] }>(withParams('/api/payments', { invoiceId: invoice.id })),
   377→  })
   378→
   379→  const [amount, setAmount] = useState('')
   380→  const [method, setMethod] = useState<PaymentMethod>('BANK')
   381→  const [reference, setReference] = useState('')
   382→  const [paidAt, setPaidAt] = useState(toDateInputValue(new Date()))
   383→
   384→  const canRecord =
   385→    isFinance && (invoice.status === 'SENT' || invoice.status === 'PARTIALLY_PAID')
   386→
   387→  const amountNum = parseFloat(amount)
   388→  const amountError =
   389→    amount.trim() === ''
   390→      ? null
   391→      : isNaN(amountNum) || amountNum <= 0
   392→        ? 'Enter an amount greater than zero'
   393→        : amountNum > invoice.balance
   394→          ? 'Amount exceeds the balance due'
   395→          : null
   396→
   397→  const recordMutation = useMutation({
   398→    mutationFn: () =>
   399→      api('/api/payments', {
   400→        method: 'POST',
   401→        body: {
   402→          invoiceId: invoice.id,
   403→          amount: amountNum,
   404→          method,
   405→          reference: reference.trim() || null,
   406→          paidAt,
   407→        },
   408→      }),
   409→    onSuccess: () => {
   410→      toast.success('Payment recorded')
   411→      void queryClient.invalidateQueries()
   412→      setAmount('')
   413→      setReference('')
   414→    },
   415→    onError: (err) =>
   416→      toast.error(err instanceof ApiError ? err.message : 'Something went wrong'),
   417→  })
   418→
   419→  function onSubmit(e: React.FormEvent) {
   420→    e.preventDefault()
   421→    if (amount.trim() === '' || amountError) return
   422→    recordMutation.mutate()
   423→  }
   424→
   425→  const payments = paymentsQuery.data?.items ?? []
   426→
   427→  return (
   428→    <Dialog open onOpenChange={(open) => !open && onClose()}>
   429→      <DialogContent className="sm:max-w-[480px]">
   430→        <DialogHeader>
   431→          <DialogTitle className="font-mono text-base">{invoice.invoiceNo}</DialogTitle>
   432→          <DialogDescription>
   433→            {invoice.projectName} · Balance due{' '}
   434→            <span className="font-medium tabular-nums text-foreground">
   435→              {formatMoney(invoice.balance)}
   436→            </span>
   437→          </DialogDescription>
   438→        </DialogHeader>
   439→
   440→        <div className="space-y-3">
   441→          <div className="max-h-64 overflow-y-auto rounded-md border">
   442→            {paymentsQuery.isLoading ? (
   443→              <div className="p-3">
   444→                <LoadingRows rows={3} />
   445→              </div>
   446→            ) : paymentsQuery.isError ? (
   447→              <ErrorState
   448→                message={paymentsQuery.error instanceof Error ? paymentsQuery.error.message : 'Failed to load payments'}
   449→                onRetry={() => void paymentsQuery.refetch()}
   450→              />
   451→            ) : payments.length === 0 ? (
   452→              <EmptyState title="No payments recorded" />
   453→            ) : (
   454→              <ul className="divide-y">
   455→                {payments.map((p) => (
   456→                  <li key={p.id} className="flex items-center justify-between gap-3 px-3 py-2.5">
   457→                    <div className="min-w-0">
   458→                      <p className="text-sm font-medium tabular-nums">{formatMoney(p.amount)}</p>
   459→                      <p className="text-xs text-muted-foreground">
   460→                        {PAYMENT_METHOD_LABELS[p.method]} · {formatDate(p.paidAt)}
   461→                      </p>
   462→                    </div>
   463→                    {p.reference ? (
   464→                      <p className="max-w-[45%] truncate font-mono text-xs text-muted-foreground">
   465→                        {p.reference}
   466→                      </p>
   467→                    ) : null}
   468→                  </li>
   469→                ))}
   470→              </ul>
   471→            )}
   472→          </div>
   473→
   474→          {canRecord ? (
   475→            <>
   476→              <Separator />
   477→              <form className="space-y-3" onSubmit={onSubmit}>
   478→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   479→                  <div className="space-y-1.5">
   480→                    <Label htmlFor="payment-amount">Amount (₹)</Label>
   481→                    <Input
   482→                      id="payment-amount"
   483→                      type="number"
   484→                      min="0"
   485→                      step="0.01"
   486→                      inputMode="decimal"
   487→                      value={amount}
   488→                      onChange={(e) => setAmount(e.target.value)}
   489→                      placeholder={String(invoice.balance)}
   490→                    />
   491→                    {amountError ? (
   492→                      <p className="text-sm text-red-600 dark:text-red-400">{amountError}</p>
   493→                    ) : null}
   494→                  </div>
   495→                  <div className="space-y-1.5">
   496→                    <Label>Method</Label>
   497→                    <Select value={method} onValueChange={(v) => setMethod(v as PaymentMethod)}>
   498→                      <SelectTrigger className="w-full" aria-label="Payment method">
   499→                        <SelectValue placeholder="Method" />
   500→                      </SelectTrigger>
   501→                      <SelectContent>
   502→                        {PAYMENT_METHODS.map((m) => (
   503→                          <SelectItem key={m} value={m}>
   504→                            {PAYMENT_METHOD_LABELS[m]}
   505→                          </SelectItem>
   506→                        ))}
   507→                      </SelectContent>
   508→                    </Select>
   509→                  </div>
   510→                  <div className="space-y-1.5">
   511→                    <Label htmlFor="payment-reference">Reference (optional)</Label>
   512→                    <Input
   513→                      id="payment-reference"
   514→                      value={reference}
   515→                      onChange={(e) => setReference(e.target.value)}
   516→                      placeholder="UTR / cheque no."
   517→                    />
   518→                  </div>
   519→                  <div className="space-y-1.5">
   520→                    <Label htmlFor="payment-date">Date</Label>
   521→                    <Input
   522→                      id="payment-date"
   523→                      type="date"
   524→                      value={paidAt}
   525→                      onChange={(e) => setPaidAt(e.target.value)}
   526→                    />
   527→                  </div>
   528→                </div>
   529→                <Button
   530→                  type="submit"
   531→                  className="w-full"
   532→                  disabled={recordMutation.isPending || amount.trim() === '' || amountError !== null}
   533→                >
   534→                  {recordMutation.isPending ? 'Recording…' : 'Record Payment'}
   535→                </Button>
   536→              </form>
   537→            </>
   538→          ) : null}
   539→        </div>
   540→      </DialogContent>
   541→    </Dialog>
   542→  )
   543→}

   549→function NewInvoiceDialog({
   550→  open,
   551→  onOpenChange,
   552→  projects,
   553→}: {
   554→  open: boolean
   555→  onOpenChange: (open: boolean) => void
   556→  projects: MetaResponse['projects']
   557→}) {
   558→  const queryClient = useQueryClient()
   559→
   560→  const [projectId, setProjectId] = useState('')
   561→  const [amount, setAmount] = useState('')
   562→  const [gstPct, setGstPct] = useState('18')
   563→  const [issueDate, setIssueDate] = useState(toDateInputValue(new Date()))
   564→  const [dueDate, setDueDate] = useState(defaultDueDate())
   565→  const [submitted, setSubmitted] = useState(false)
   566→
   567→  function resetForm() {
   568→    setProjectId('')
   569→    setAmount('')
   570→    setGstPct('18')
   571→    setIssueDate(toDateInputValue(new Date()))
   572→    setDueDate(defaultDueDate())
   573→    setSubmitted(false)
   574→  }
   575→
   576→  function close() {
   577→    resetForm()
   578→    onOpenChange(false)
   579→  }
   580→
   581→  const amountNum = parseFloat(amount)
   582→  const amountError =
   583→    amount.trim() === ''
   584→      ? null
   585→      : isNaN(amountNum) || amountNum <= 0
   586→        ? 'Enter an amount greater than zero'
   587→        : null
   588→  const dueDateError = dueDate <= issueDate ? 'Due date must be after the issue date' : null
   589→  const valid = projectId !== '' && amountError === null && amount.trim() !== '' && dueDateError === null
   590→
   591→  const createMutation = useMutation({
   592→    mutationFn: () =>
   593→      api<{ invoice: { id: string; invoiceNo: string } }>('/api/invoices', {
   594→        method: 'POST',
   595→        body: { projectId, amount: amountNum, gstPct: Number(gstPct), issueDate, dueDate },
   596→      }),
   597→    onSuccess: (data) => {
   598→      toast.success(`Invoice ${data.invoice.invoiceNo} created as draft`)
   599→      void queryClient.invalidateQueries()
   600→      close()
   601→    },
   602→    onError: (err) =>
   603→      toast.error(err instanceof ApiError ? err.message : 'Something went wrong'),
   604→  })
   605→
   606→  function onSubmit(e: React.FormEvent) {
   607→    e.preventDefault()
   608→    setSubmitted(true)
   609→    if (!valid) return
   610→    createMutation.mutate()
   611→  }
   612→
   613→  return (
   614→    <Dialog
   615→      open={open}
   616→      onOpenChange={(next) => {
   617→        if (!next) close()
   618→      }}
   619→    >
   620→      <DialogContent className="sm:max-w-[480px]">
   621→        <DialogHeader>
   622→          <DialogTitle>New Invoice</DialogTitle>
   623→          <DialogDescription>
   624→            The invoice is created as a draft — send it to the client once you are ready to collect.
   625→          </DialogDescription>
   626→        </DialogHeader>
   627→        <form className="space-y-3" onSubmit={onSubmit}>
   628→          <div className="space-y-1.5">
   629→            <Label>Project</Label>
   630→            <Select value={projectId} onValueChange={setProjectId}>
   631→              <SelectTrigger className="w-full" aria-label="Project">
   632→                <SelectValue placeholder="Select project" />
   633→              </SelectTrigger>
   634→              <SelectContent>
   635→                {projects.map((project) => (
   636→                  <SelectItem key={project.id} value={project.id}>
   637→                    {project.name}
   638→                  </SelectItem>
   639→                ))}
   640→              </SelectContent>
   641→            </Select>
   642→            {submitted && projectId === '' ? (
   643→              <p className="text-sm text-red-600 dark:text-red-400">Select a project</p>
   644→            ) : null}
   645→          </div>
   646→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   647→            <div className="space-y-1.5">
   648→              <Label htmlFor="invoice-amount">Amount (₹)</Label>
   649→              <Input
   650→                id="invoice-amount"
   651→                type="number"
   652→                min="0"
   653→                step="0.01"
   654→                inputMode="decimal"
   655→                value={amount}
   656→                onChange={(e) => setAmount(e.target.value)}
   657→              />
   658→              {submitted && amountError ? (
   659→                <p className="text-sm text-red-600 dark:text-red-400">{amountError}</p>
   660→              ) : null}
   661→            </div>
   662→            <div className="space-y-1.5">
   663→              <Label>GST</Label>
   664→              <Select value={gstPct} onValueChange={setGstPct}>
   665→                <SelectTrigger className="w-full" aria-label="GST percentage">
   666→                  <SelectValue placeholder="GST" />
   667→                </SelectTrigger>
   668→                <SelectContent>
   669→                  {GST_OPTIONS.map((g) => (
   670→                    <SelectItem key={g} value={String(g)}>
   671→                      {g}%
   672→                    </SelectItem>
   673→                  ))}
   674→                </SelectContent>
   675→              </Select>
   676→            </div>
   677→            <div className="space-y-1.5">
   678→              <Label htmlFor="invoice-issue-date">Issue date</Label>
   679→              <Input
   680→                id="invoice-issue-date"
   681→                type="date"
   682→                value={issueDate}
   683→                onChange={(e) => setIssueDate(e.target.value)}
   684→              />
   685→            </div>
   686→            <div className="space-y-1.5">
   687→              <Label htmlFor="invoice-due-date">Due date</Label>
   688→              <Input
   689→                id="invoice-due-date"
   690→                type="date"
   691→                value={dueDate}
   692→                min={issueDate}
   693→                onChange={(e) => setDueDate(e.target.value)}
   694→              />
   695→              {submitted && dueDateError ? (
   696→                <p className="text-sm text-red-600 dark:text-red-400">{dueDateError}</p>
   697→              ) : null}
   698→            </div>
   699→          </div>
   700→          <DialogFooter>
   701→            <Button type="button" variant="outline" onClick={close}>
   702→              Cancel
   703→            </Button>
   704→            <Button type="submit" disabled={createMutation.isPending}>
   705→              {createMutation.isPending ? 'Creating…' : 'Create Invoice'}
   706→            </Button>
   707→          </DialogFooter>
   708→        </form>
   709→      </DialogContent>
   710→    </Dialog>
   711→  )
   712→}
